## 設定 [#setup]

此外掛適用於 React Router、TanStack Start、Waku 及其他 Vite 應用程式。Next.js 有[專屬的設定方式](/docs/mdx/next)。Astro 則是搭配 [`@fumadocs/astro`](/docs/astro/manual-installation) 使用 Astro 本身的內容集合，而非 Fumadocs MDX。

<div className="fd-steps [&_h3]:fd-step">
  ### 安裝 [#installation]

  ```npm
  npm i fumadocs-mdx fumadocs-core @types/mdx
  ```

  加入 Vite 外掛：

  ```ts tab="React Router" title="vite.config.ts"
  import { defineConfig } from 'vite';
  import { reactRouter } from '@react-router/dev/vite';
  import { fumadocsMdx } from 'fumadocs-mdx/vite';

  export default defineConfig({
    plugins: [
      fumadocsMdx(), // [!code ++]
      reactRouter(),
    ],
  });
  ```

  ```ts tab="TanStack Start" title="vite.config.ts"
  import { defineConfig } from 'vite';
  import { tanstackStart } from '@tanstack/react-start/plugin/vite';
  import react from '@vitejs/plugin-react';
  import { fumadocsMdx } from 'fumadocs-mdx/vite';

  export default defineConfig({
    plugins: [
      fumadocsMdx(), // [!code ++]
      tanstackStart(),
      react(),
    ],
  });
  ```

  ```ts tab="Waku" title="waku.config.ts"
  import { type Config, defineConfig } from 'waku/config';
  import { fumadocsMdx } from 'fumadocs-mdx/vite';
  import type { UserConfig } from 'vite';

  export default defineConfig({
    vite: {
      plugins: [
        // [!code ++]
        fumadocsMdx(),
      ],
      resolve: {
        tsconfigPaths: true,
      },
    } satisfies UserConfig as Config['vite'],
  });
  ```

  ### 與 Fumadocs 整合 [#integrate-with-fumadocs]

  建立 `lib/source.ts` 檔案：

  ```ts title="lib/source.ts"
  import { defineDocs } from 'fumadocs-mdx/macro';
  import { loader } from 'fumadocs-core/source';

  const docs = defineDocs({
    dir: 'content/docs',
  });

  export const source = loader({
    baseUrl: '/docs',
    source: docs.toFumadocsSource(),
  });
  ```

  ### 完成 [#done]

  現在您可以在 `content/docs` 資料夾中撰寫內容了。
</div>
