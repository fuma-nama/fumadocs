## インストール [#installation]

```npm
npm install takumi-js
```

<show when={props.framework === 'next'}>
  `takumi-js` は Node.js 上でネイティブバインディング (`@takumi-rs/core`) を読み込みます。サーバーバンドルに含まれないように、外部パッケージとして指定してください。

  ```ts title="next.config.ts"
  import type { NextConfig } from 'next';

  const config: NextConfig = {
    // [!code ++]

  };
  ```

## メタデータ画像 [#metadata-image]

ページの画像 URL を返す関数を追加し、ドキュメントページのメタデータからその画像を参照します。

<show when={props.framework === 'next'}>
  ```ts tab="lib/source.ts"
  // [!code ++:8]
  export function getPageImageUrl(page: (typeof source)['$inferPage']) {
    const segments = [...page.slugs, 'image.webp'];

    return {
      segments,
      url: '/' + [page.locale, 'og', 'docs', ...segments].filter(Boolean).join('/'),
    };
  }
  ```

  ```tsx tab="app/docs/[[...slug]]/page.tsx"
  import { notFound } from 'next/navigation';
  import { source, getPageImageUrl } from '@/lib/source';
  import type { Metadata } from 'next';

  export async function generateMetadata(props: PageProps<'/docs/[[...slug]]'>): Promise<Metadata> {
    const params = await props.params;
    const page = source.getPage(params.slug);
    if (!page) notFound();

    return {
      title: page.data.title,
      description: page.data.description,
      openGraph: {
        // [!code ++]
        images: getPageImageUrl(page).url,
      },
    };
  }
  ```
</show>

> `/docs/my-page` の画像は `/og/docs/my-page/image.webp` に配置されます。WebP 画像は PNG 画像よりもファイルサイズが小さいためです。PNG を生成したい場合は、代わりに `image.png` と `format: 'png'` を使用してください。

次に、画像を生成するルートを作成します。

<show when={props.framework === 'next'}>
  ```tsx title="app/og/docs/[...slug]/route.tsx"
  import { getPageImageUrl, source } from '@/lib/source';
  import { notFound } from 'next/navigation';
  import { generateOGImage } from 'fumadocs-ui/og/takumi';

  export const revalidate = false;

  export async function GET(_req: Request, { params }: RouteContext<'/og/docs/[...slug]'>) {
    const { slug } = await params;
    const page = source.getPage(slug.slice(0, -1));
    if (!page) notFound();

    return generateOGImage({
      title: page.data.title,
      description: page.data.description,
      site: 'My App',
      format: 'webp',
    });
  }

  export function generateStaticParams() {
    return source.getPages().map((page) => ({
      lang: page.locale,
      slug: getPageImageUrl(page).segments,
    }));
  }
  ```
</show>

<show when={props.framework === 'react-router'}>
  ```tsx tab="app/routes/docs.og.tsx"
  import { source } from '@/lib/source';
  import { generateOGImage } from 'fumadocs-ui/og/takumi';
  import type { Route } from './+types/docs.og';

  export function loader({ params }: Route.LoaderArgs) {
    const slugs = params['*']
      .split('/')
      .filter((v) => v.length > 0)
      .slice(0, -1);
    const page = source.getPage(slugs);

    if (!page) throw new Response(undefined, { status: 404 });

    return generateOGImage({
      title: page.data.title,
      description: page.data.description,
      site: 'My App',
      format: 'webp',
    });
  }
  ```

  ```ts tab="app/routes.ts"
  import { type RouteConfig, route } from '@react-router/dev/routes';

  export default [
    // [!code ++]
    route('/og/docs/*', 'routes/docs.og.tsx'),
  ] satisfies RouteConfig;
  ```

  ```ts tab="react-router.config.ts"
  import type { Config } from '@react-router/dev/config';
  import { glob } from 'node:fs/promises';
  import { createGetUrl, getSlugs } from 'fumadocs-core/source';
  import { getPageImagePath } from './app/lib/shared';

  const getUrl = createGetUrl('/docs');

  export default {
    ssr: true,
    // [!code ++:12]
    async prerender({ getStaticPaths }) {
      const paths = [...getStaticPaths()];

      for await (const entry of glob('**/*.{md,mdx}', { cwd: 'content/docs' })) {
        const slugs = getSlugs(entry);

        paths.push(getUrl(slugs));
        paths.push(getPageImagePath(slugs));
      }

      return paths;
    },
  } satisfies Config;
  ```
</show>

  ```ts tab="src/routes/og/docs/$.ts"
  import { createFileRoute } from '@tanstack/react-router';
  import { generateOGImage } from 'fumadocs-ui/og/takumi';
  import { source } from '@/lib/source';

  export const Route = createFileRoute('/og/docs/$')({
    server: {
      handlers: {
        GET: ({ params }) => {

          if (!page) return new Response(undefined, { status: 404 });

          return generateOGImage({
            title: page.data.title,
            description: page.data.description,
            site: 'My App',
            format: 'webp',
          });
        },
      },
    },
  });
  ```

  ```ts tab="vite.config.ts"
  // SPA モードの場合のみ: TanStack Start はレスポンスをテキストとしてプリレンダリングするため、画像が破損します
  import { defineConfig } from 'vite';
  import { nitro } from 'nitro/vite';
  // [!code ++:3]
  import { glob } from 'node:fs/promises';
  import { getSlugs } from 'fumadocs-core/source';
  import { getPageImageUrl } from './src/lib/shared';

  // [!code ++:4]
  const imageRoutes: string[] = [];
  for await (const entry of glob('**/*.{md,mdx}', { cwd: 'content/docs' })) {
    imageRoutes.push(getPageImageUrl({ slugs: getSlugs(entry) }).url);
  }

  export default defineConfig({
    plugins: [
      // その他のプラグイン...
      nitro({
        // [!code ++]
        prerender: { routes: imageRoutes },
      }),
    ],
  });
  ```

<show when={props.framework === 'waku'}>
  ```tsx title="src/pages/_api/og/docs/[...slugs]/image.webp.tsx"
  import { source } from '@/lib/source';
  import { generateOGImage } from 'fumadocs-ui/og/takumi';
  import { ApiContext } from 'waku/router';

  export async function GET(_: Request, { params }: ApiContext<'/og/docs/[...slugs]/image.webp'>) {
    const page = source.getPage(params.slugs);

    if (!page) return new Response(undefined, { status: 404 });

    return generateOGImage({
      title: page.data.title,
      description: page.data.description,
      site: 'My App',
      format: 'webp',
    });
  }

  export async function getConfig() {
    const pages = source
      .generateParams()
      .map((item) => (item.lang ? [item.lang, ...item.slug] : item.slug));

    return {
      render: 'static' as const,
      staticPaths: pages,
    } as const;
  }
  ```

> Takumi には全ウェイト (100-900) に対応した [`Geist`](https://vercel.com/font) フォントと `Geist Mono` フォントが同梱されているため、フォントの設定は不要です。

高度な使い方については、[Takumi のドキュメント](https://takumi.kane.tw/docs)を参照してください。
