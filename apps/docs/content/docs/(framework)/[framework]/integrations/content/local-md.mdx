## Introduction

`@fumadocs/local-md` is a content source for local Markdown/MDX files. It is bundleless by design, so it works fully at runtime.

Compared to [MDX Remote](/docs/integrations/content/mdx-remote), it handles more of the pipeline (frontmatter, meta files, search indexing) while focused on local files.

<show when={props.framework !== 'astro'}>

Compared to [Fumadocs MDX](/docs/mdx), it doesn't need a type-gen or bundler to work, but build-time image optimization is disabled.

</show>

<show when={props.framework === 'astro'}>

Compared to Astro content collections, it compiles your content at runtime instead of through the bundler, so build-time image optimization is disabled.

</show>

### Limitations

- No build-time image optimization.
- No imports/exports in MDX files, but you can pass components when rendering.

## Setup

Install the package:

```package-install
@fumadocs/local-md shiki
```

> `shiki` is installed because it has to be externalized by the bundler.

Create a `localMd` instance and connect it to Fumadocs:

```ts title="lib/source.ts"
import { dynamicLoader } from 'fumadocs-core/source';
import { localMd } from '@fumadocs/local-md';

const docs = localMd({
  dir: 'content/docs',
  // options
});

const docsLoader = dynamicLoader(docs.dynamicSource(), {
  baseUrl: '/docs',
});

export async function getSource() {
  return docsLoader.get();
}
```

### Schema

You may pass `frontmatterSchema` and `metaSchema` to customize the validation schemas:

```ts title="lib/source.ts"
import { localMd } from '@fumadocs/local-md';
import { pageSchema, metaSchema } from 'fumadocs-core/source/schema';

const docs = localMd({
  dir: 'content/docs',
  frontmatterSchema: pageSchema.extend({
    // ...
  }),
  metaSchema: metaSchema.extend({
    // ...
  }),
});
```

## Usage

`getSource()` returns a normal [content loader](/docs/headless/source-api). Each page has `page.data.load()`, which compiles the page and returns a renderer.

```tsx tab="app/docs/layout.tsx"
import { getSource } from '@/lib/source';
import { DocsLayout } from 'fumadocs-ui/layouts/docs';

export default async function Layout({ children }: LayoutProps<'/docs'>) {

  return <DocsLayout tree={source.getPageTree()}>{children}</DocsLayout>;
```

```tsx tab="app/docs/[[...slug]]/page.tsx"
import { notFound } from 'next/navigation';
import { DocsBody, DocsPage, DocsTitle } from 'fumadocs-ui/layouts/docs/page';
import { getSource } from '@/lib/source';
import { getMDXComponents } from '@/components/mdx';

export default async function Page(props: PageProps<'/docs/[[...slug]]'>) {
  const params = await props.params;
  const source = await getSource();
  const page = source.getPage(params.slug);
  if (!page) notFound();

  const { render } = await page.data.load();
  const { body, toc } = await render(getMDXComponents());

  return (
    <DocsPage toc={toc}>
      <DocsTitle>{page.data.title}</DocsTitle>
      <DocsBody>{body}</DocsBody>
    </DocsPage>
  );
}
```

<show when={props.framework === 'react-router'}>

```tsx title="app/routes/docs.tsx"
import type { Route } from './+types/docs';
import { useMemo } from 'react';
import { DocsLayout } from 'fumadocs-ui/layouts/docs';
import { DocsBody, DocsPage, DocsTitle } from 'fumadocs-ui/layouts/docs/page';
import { useFumadocsLoader } from 'fumadocs-core/source/client';
import { rendererFromSerialized } from '@fumadocs/local-md/client';
import { getSource } from '@/lib/source';
import { useMDXComponents } from '@/components/mdx';

export async function loader({ params }: Route.LoaderArgs) {
  const source = await getSource();
  const page = source.getPage(params['*'].split('/').filter((v) => v.length > 0));
  if (!page) throw new Response('Not found', { status: 404 });

  const { serialize } = await page.data.load();
  return {
    title: page.data.title,
    // the syntax tree, rendered on the client
    render: serialize(),
    pageTree: await source.serializePageTree(source.getPageTree()),
  };
}

export default function Page({ loaderData }: Route.ComponentProps) {
  const { title, render, pageTree } = useFumadocsLoader(loaderData);
  const renderer = useMemo(() => rendererFromSerialized(render), [render]);
  const { body, toc } = renderer.renderSync(useMDXComponents());

  return (
    <DocsLayout tree={pageTree}>
      <DocsPage toc={toc}>
        <DocsTitle>{title}</DocsTitle>
        <DocsBody>{body}</DocsBody>
      </DocsPage>
    </DocsLayout>
  );
}
```

```tsx title="src/routes/docs/$.tsx"
import { useMemo } from 'react';
import { DocsLayout } from 'fumadocs-ui/layouts/docs';
import { DocsBody, DocsPage, DocsTitle } from 'fumadocs-ui/layouts/docs/page';
import { useFumadocsLoader } from 'fumadocs-core/source/client';
import { rendererFromSerialized } from '@fumadocs/local-md/client';
import type { MarkdownRendererSerializedOptions } from '@fumadocs/local-md';
import { getSource } from '@/lib/source';
import { useMDXComponents } from '@/components/mdx';

// the type-level check of server function results can't verify the syntax tree,
// register it as serializable (install `@tanstack/router-core` if it isn't resolvable)
declare module '@tanstack/router-core' {
  interface SerializableExtensions {
    fumadocsMarkdownRenderer: MarkdownRendererSerializedOptions;
  }

export const Route = createFileRoute('/docs/$')({
  component: Page,
  loader: ({ params }) => serverLoader({ data: params._splat?.split('/') ?? [] }),
});

const serverLoader = createServerFn({
  method: 'GET',
})
  .validator((slugs: string[]) => slugs)
  .handler(async ({ data: slugs }) => {
    const source = await getSource();
    const page = source.getPage(slugs);
    if (!page) throw notFound();

    const { serialize } = await page.data.load();
    return {
      title: page.data.title,
      // the syntax tree, rendered on the client
      render: serialize(),
      pageTree: await source.serializePageTree(source.getPageTree()),
    };
  });

function Page() {

  const renderer = useMemo(() => rendererFromSerialized(render), [render]);
  const { body, toc } = renderer.renderSync(useMDXComponents());

  return (
    <DocsLayout tree={pageTree}>
      <DocsPage toc={toc}>
        <DocsTitle>{title}</DocsTitle>
        <DocsBody>{body}</DocsBody>
      </DocsPage>
    </DocsLayout>
  );
}
```

```tsx tab="src/components/docs.tsx"
import { useMemo } from 'react';
import { type SerializedPageTree, useFumadocsLoader } from 'fumadocs-core/source/client';
import { DocsLayout } from 'fumadocs-ui/layouts/docs';
import { DocsBody, DocsPage, DocsTitle } from 'fumadocs-ui/layouts/docs/page';
import { rendererFromSerialized } from '@fumadocs/local-md/client';
import type { MarkdownRendererSerializedOptions } from '@fumadocs/local-md';
import { Provider, type ProviderProps } from '@/components/provider';
import { getMDXComponents } from '@/components/mdx';

export interface DocsProps extends ProviderProps {
  tree: SerializedPageTree;
  title: string;
  // the syntax tree, rendered in the island
  render: MarkdownRendererSerializedOptions;
}

export function Docs({ tree, title, render, ...props }: DocsProps) {
  const data = useFumadocsLoader({ tree });
  const renderer = useMemo(() => rendererFromSerialized(render), [render]);
  const { body, toc } = renderer.renderSync(getMDXComponents());

  return (
    <Provider {...props}>
      <DocsLayout tree={data.tree}>
        <DocsPage toc={toc}>
          <DocsTitle>{title}</DocsTitle>
          <DocsBody>{body}</DocsBody>
        </DocsPage>
      </DocsLayout>
    </Provider>
  );
}
```

### Include

[`<include>`](/docs/mdx/include) is enabled by default:

```mdx title="content/docs/page.mdx"
<include>../shared/note.md</include>
```

In Markdown files, use the `::include[../shared/note.md]` directive with `remark-directive` configured:

```ts title="lib/source.ts"
import remarkDirective from 'remark-directive';

const docs = localMd({
  dir: 'content/docs',
  mdOptions: { remarkPlugins: [remarkDirective] },
});
```

### Hot Reload

`@fumadocs/local-md` reads files at runtime, so nothing watches them by default.

<show when={props.framework === 'next'}>

During development, run the bundled dev server alongside your app. The dev server watches files in a single process and broadcasts changes over a websocket, so it works when Next.js runs multiple workers.

```json title="package.json"
{
  "scripts": {
    "dev": "local-md dev -- next dev"
  }
}
```

Connect the source to the dev server:

```ts title="lib/source.ts"
import { watchWithDevServer } from '@fumadocs/local-md/dev/ws';

if (process.env.NODE_ENV === 'development') {
  void watchWithDevServer(docs);
}
```

The command publishes its URL through an environment variable, so no configuration is needed. Pass one explicitly if you run the server yourself, and use `-p` to change the port:

```ts
void watchWithDevServer(docs, { url: 'ws://127.0.0.1:8000/_fumadocs_local_md' });
```

This invalidates the source on the server, while Next.js keeps the rendered page in the browser. Render `<DevClient />` in your root layout so the page refreshes too:

```tsx title="app/layout.tsx"
import { DevClient } from '@fumadocs/local-md/dev/react-client';

export function RootLayout({ children }) {
  return (
    <RootProvider>
      {/* [!code ++] */}
      <DevClient />
      {children}
    </RootProvider>
  );
}
```

</show>

The source reloads when Markdown or MDX files change, including files they include. To wire up your own watcher, watch `docs.dir` and the files passed to `docs.onDependency()`, then call `docs.invalidateFile(path)` when a file changes.

### Nitro

When you deploy with Nitro, make two changes in the Vite config:

```ts title="vite.config.ts"
import { defineConfig } from 'vite';
import { nitro } from 'nitro/vite';

export default defineConfig({
  plugins: [
    // Nitro can't bundle the Wasm imports of Shiki, its default entry inlines the binary
    nitro({ wasm: false }),
  ],
  resolve: {
    alias: [
      // Nitro keeps the `require('react')` of this shim, which loads a second React at runtime
      { find: /^use-sync-external-store\/shim$/, replacement: 'react' },
    ],
  },
});
```

</show>

### JavaScript Engine

MDX files (`*.mdx`) are executed with `new Function`, so they do not work where `eval()` is disallowed (for example, Cloudflare Workers).

Markdown files (`*.md`) render from the syntax tree without `eval()`. A built-in virtual JavaScript engine evaluates JSX generated by plugins. Replace it with `rendererOptions.executor`.

<Callout type="warn">

Do not put untrusted MDX files in `dir`, they run as JavaScript in your app.

</Callout>

### Disable Revalidation

`staticSource()` returns a one-time snapshot with no revalidation. It works with `loader()` instead of `dynamicLoader()`:

```ts title="lib/source.ts"
import { loader } from 'fumadocs-core/source';
import { localMd } from '@fumadocs/local-md';

const docs = localMd({
  dir: 'content/docs',
});

export const source = loader(await docs.staticSource(), {
  baseUrl: '/docs',
});
```

The page data API also changes:

```tsx tab="Before"
const page = source.getPage(['...']);

// title & description
page.data.title;

// custom frontmatter properties
page.data.full;

// getText() API
await page.data.getText('processed');

// compiled properties:
page.data.structuredData;
page.data.toc;
return (
  <div>
    <page.data.body components={{}} />
  </div>
);
```

```tsx tab="After"
const page = docs.getPage(['...']);

// title & description (unchanged)
page.data.title;

// custom frontmatter properties
page.data.frontmatter.full;

// there's no getText() API, use data.content instead
console.log(page.data.content);

// compiled properties:
const structuredData = await page.data.structuredData();
const { toc, body } = await (await page.data.load()).render(mdxComponents);

return <div>{body}</div>;
```
