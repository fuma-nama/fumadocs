## Overview

Configure i18n routing on both {frameworkName(props.framework)} and Fumadocs.

Fumadocs handles i18n for the docs content and sidebar. Internationalize the rest of your app with your own i18n setup.

<show when={props.framework === 'next'}>

You can also use libraries like [next-intl](https://github.com/amannn/next-intl) with Fumadocs. To learn the basics, see [i18n in Next.js](https://nextjs.org/docs/app/building-your-application/routing/internationalization).

</show>

```ts title="app/routes.ts"
import { index, route, type RouteConfig } from '@react-router/dev/routes'; // [!code --]
import { route, type RouteConfig } from '@react-router/dev/routes'; // [!code ++]

export default [
  index('routes/home.tsx'), // [!code --]
  route('docs/*', 'routes/docs.tsx'), // [!code --]
  // [!code ++:2]
  route(':lang', 'routes/home.tsx'),
  route(':lang/docs/*', 'routes/docs.tsx'),
  route('api/search', 'routes/search.ts'),
] satisfies RouteConfig;
```

To serve the default language without a prefix (e.g. `/docs`), make the parameter optional and enable `hideLocale`:

```ts tab="app/routes.ts"
export default [
  // [!code highlight:2]
  route(':lang?', 'routes/home.tsx'),
  route(':lang?/docs/*', 'routes/docs.tsx'),
  route('api/search', 'routes/search.ts'),
] satisfies RouteConfig;
```

### Translations

Define your translations in `lib/layout.shared.tsx`. See [Translations](/docs/ui/translations) for more details, including the official language packs.

You can also add a `locale` parameter to `baseOptions()` for localized layout options:

<show when={props.framework === 'next'}>

```tsx tab="src/components/provider.tsx"
  children?: ReactNode;

export function Provider({ lang, pathname, params, children }: ProviderProps) {
  return (
    <RootProvider
      pathname={pathname}
      params={params}
      navigate={navigate}
```

```tsx tab="Docs Page" title="src/pages/[lang]/docs/[...slugs].tsx"
    .map((item) => (item.lang ? [item.lang, ...item.slug] : item.slug)); // [!code highlight]

  return {
    render: 'static',
    staticPaths: pages,
  } as const;
}
```

### Navigation

Fumadocs only handles navigation for its own layouts (e.g. sidebar).
