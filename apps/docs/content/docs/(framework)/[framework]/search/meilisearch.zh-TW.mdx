## 設定 [#setup]

Fumadocs CLI 可以幫您完成設定，包括建立搜尋對話方塊、`static.json` 路由，以及在每次建置後同步搜尋索引的指令碼：

```package-install
npx @fumadocs/cli feature search --provider meilisearch
```

或手動設定：

1. 整合 [Meilisearch](/docs/headless/search/meilisearch)。

2. 建立搜尋對話方塊，並將 `host` 和 `apiKey` 替換為您的主機 URL 與搜尋 API 金鑰。

### 標籤篩選 [#tag-filter]

您也可以視需要加入依標籤篩選結果的 UI。請先在文件上設定[標籤篩選](/docs/headless/search/meilisearch#tag-filter)，再加入以下內容：

```tsx
'use client';

import {
  SearchDialog,
  SearchDialogContent,
  SearchDialogFooter,
  SearchDialogOverlay,
  type SharedProps,
  TagsList,
  TagsListItem,
} from 'fumadocs-ui/components/dialog/search';
import { useState } from 'react';
import { useMeilisearch } from 'fumadocs-core/search/client';

export default function CustomSearchDialog(props: SharedProps) {
  // [!code ++]
  const [tag, setTag] = useState<string | undefined>();
  const search = useMeilisearch({
    tag, // [!code ++]
    // 其他選項……
  });

  return (
    <SearchDialog {...search} {...props}>
      <SearchDialogOverlay />
      <SearchDialogContent>
        ...
        <SearchDialogFooter className="flex flex-row">
          {/* [!code ++:3] */}
          <TagsList tag={tag} onTagChange={setTag}>
            <TagsListItem value="my-value">My Value</TagsListItem>
          </TagsList>
        </SearchDialogFooter>
      </SearchDialogContent>
    </SearchDialog>
  );
}
```
