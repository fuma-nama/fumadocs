## Getting Started

Install the packages:

<show when={props.framework !== 'astro'}>

```npm
npm i fumadocs-core fumadocs-ui@npm:@fumadocs/base-ui
```

</show>

<show when={props.framework === 'astro'}>

```npm
npm i fumadocs-core fumadocs-ui@npm:@fumadocs/base-ui @fumadocs/astro @astrojs/react react react-dom @types/mdx
```

</show>

Like the templates, the command installs `fumadocs-ui` as an alias of `@fumadocs/base-ui`, the [Base UI](/docs/ui/component-library) version of Fumadocs UI. For Radix UI, install `fumadocs-ui` without the alias.

### Styles

Add the following to your Tailwind CSS file:

```css
@import 'tailwindcss';
/* [!code ++:2] */
@import 'fumadocs-ui/css/neutral.css';
@import 'fumadocs-ui/css/preset.css';
```

<show when={props.framework === 'next'}>

> Fumadocs UI has no default font. Pick one from `next/font`.

</show>

### Root Provider

</show>

<show when={props.framework === 'next' || props.framework === 'react-router' || props.framework === 'tanstack-start'}>

Wrap your entire app inside [Root Provider](/docs/ui/layouts/root-provider), and add the required styles to `body`.

</show>

<show when={props.framework === 'next'}>

```tsx title="app/layout.tsx"
import { RootProvider } from 'fumadocs-ui/provider/next';
import type { ReactNode } from 'react';

export default function Layout({ children }: { children: ReactNode }) {
  return (
    <html lang="en" suppressHydrationWarning>
      <body
        // required styles [!code ++:2]
        className="flex flex-col min-h-screen"
      >
        <RootProvider>{children}</RootProvider>
      </body>
    </html>
  );
}
```

</show>

<show when={props.framework === 'react-router'}>

```tsx title="app/root.tsx"
import { Links, Meta, Scripts, ScrollRestoration } from 'react-router';
import { RootProvider } from 'fumadocs-ui/provider/react-router';
import './app.css';

export function Layout({ children }: { children: React.ReactNode }) {
  return (
    <html lang="en" suppressHydrationWarning>
      <head>
        <meta charSet="utf-8" />
        <meta name="viewport" content="width=device-width, initial-scale=1" />
        <Meta />
        <Links />
      </head>
      <body className="flex flex-col min-h-screen">
        {/* [!code ++] */}
        <RootProvider>{children}</RootProvider>
        <ScrollRestoration />
        <Scripts />
      </body>
    </html>
  );
}
```

</show>

<show when={props.framework === 'tanstack-start'}>

```tsx title="src/routes/__root.tsx"
import { createRootRoute, HeadContent, Outlet, Scripts } from '@tanstack/react-router';
import * as React from 'react';
import { RootProvider } from 'fumadocs-ui/provider/tanstack';

export const Route = createRootRoute({
  component: RootComponent,
});

function RootComponent() {
  return (
    <RootDocument>
      <Outlet />
    </RootDocument>
  );
}

function RootDocument({ children }: { children: React.ReactNode }) {
  return (
    <html suppressHydrationWarning>
      <head>
        <HeadContent />
      </head>
      <body className="flex flex-col min-h-screen">
        {/* [!code ++] */}
        <RootProvider>{children}</RootProvider>
        <Scripts />
      </body>
    </html>
  );
}
```

### Integration

Add the Fumadocs integration to your Astro config:

```js title="astro.config.mjs"
import { defineConfig } from 'astro/config';
import tailwindcss from '@tailwindcss/vite';
import fumadocs from '@fumadocs/astro';

export default defineConfig({
  integrations: [fumadocs({ components: './src/components/mdx.tsx' })],
  vite: {
    plugins: [tailwindcss()],
  },
});
```
