## Overview

`fumadocs-openapi/headless` provides the runtime of API pages: the document, the selected server, the auth state of playground, the details of each operation, its example requests and the code they generate.

The built-in UI and the components installed with Fumadocs CLI render from the same hooks, so you can replace any part of the UI without re-implementing its logic.

## Page

`<OpenAPIProvider />` provides the runtime to a page, it takes the bundled document and the components the UI renders through.

```tsx title="components/api-page.tsx"
'use client';
import {
  type CodeBlockProps,
  type OpenAPIComponents,
  OpenAPIProvider,
  useOpenAPI,
} from 'fumadocs-openapi/headless';
import type { OpenAPIPageProps_Spec } from 'fumadocs-openapi/ui';
import type { HttpMethods } from 'fumadocs-openapi';
import { Schema } from '@fumadocs/api-docs/components/schema';
import { Operation } from '@/components/api/operation';

function Markdown({ md }: { md: string }) {
  return <p>{md}</p>;
}

function CodeBlock({ lang, code }: CodeBlockProps) {
  return (
    <pre>
      <code className={`language-${lang}`}>{code}</code>
    </pre>
  );
}

const components: OpenAPIComponents = {
  Markdown,
  CodeBlock,
  Heading({ depth, ...props }) {
    const As = `h${depth}` as 'h2';
    return <As {...props} />;
  },
  SchemaUI: (props) => (
    <Schema
      {...props}
      renderMarkdown={(md) => <Markdown md={md} />}
      renderCodeblock={(props) => <CodeBlock {...props} />}
    />
  ),
};

export function OpenAPIPage({ payload, operations = [] }: OpenAPIPageProps_Spec) {
  return (
    <OpenAPIProvider document={payload.bundled} proxyUrl={payload.proxyUrl} components={components}>
      {operations.map((item) => (
        <PageOperation key={`${item.method} ${item.path}`} {...item} />
      ))}
    </OpenAPIProvider>
  );
}

function PageOperation({ path, method }: { path: string; method: HttpMethods }) {
  const { document } = useOpenAPI();
  const pathItem = document.resolve(document.dereferenced.paths![path])!;

  return (
    <Operation path={path} method={method} pathItem={pathItem} operation={pathItem[method]!} />
  );
}
```

The `Operation` component can be the one installed with Fumadocs CLI, or one built from scratch as below.

| Hook                                        | Description                                   |
| ------------------------------------------- | --------------------------------------------- |
| `useOpenAPI()`                              | The document and request options of the page. |
| `useComponents()`                           | The components passed to the provider.        |
| `useServer()`                               | The selected server and its variables.        |
| `useAuth()`                                 | The auth state of playground.                 |
| `useTypeScriptDefinitions(schema, options)` | TypeScript definitions of a JSON schema.      |

## Operation

`<OperationProvider />` derives the details of an operation or webhook, and holds its selected example request. Edits from the playground update the request data, and the generated code usages follow.

```tsx title="components/api/operation.tsx"
'use client';
import type { OperationProps } from 'fumadocs-openapi/ui';
import {
  OperationProvider,
  useCodeUsage,
  useCodeUsages,
  useOperation,
  useResponseExamples,
} from 'fumadocs-openapi/headless';

export function Operation({ path, method, operation, pathItem }: OperationProps) {
  return (
    <OperationProvider path={path} method={method} operation={operation} pathItem={pathItem}>
      <Content />
    </OperationProvider>
  );
}

function Content() {
  const { title, parameters } = useOperation();
  const codeUsages = useCodeUsages();
  const responses = useResponseExamples();

  return (
    <>
      <h2>{title}</h2>
      {parameters.map(({ in: location, items }) => (
        <section key={location}>{items.map((param) => param.name).join(', ')}</section>
      ))}
      {codeUsages.map((item) => (
        <CodeUsage key={item.id} id={item.id} />
      ))}
      {responses.map((tab) => (
        <pre key={tab.code}>{JSON.stringify(tab.examples?.[0]?.sample)}</pre>
      ))}
    </>
  );
}

function CodeUsage({ id }: { id: string }) {
  const code = useCodeUsage(id);
  return code && <pre>{code}</pre>;
}
```

| Hook                                  | Description                                                                                                         |
| ------------------------------------- | ------------------------------------------------------------------------------------------------------------------- |
| `useOperation()`                      | The operation and its resolved details: `title`, `requestBody`, `parameters`, `security`, `responses`, `callbacks`. |
| `useExampleRequests()`                | The example requests and the selected one.                                                                          |
| `useExampleRequest()`                 | Request data of the selected example, following updates from the playground.                                        |
| `useCodeUsages()`, `useCodeUsage(id)` | The code usage generators, and the code they generate for the selected example.                                     |
| `useResponseExamples()`               | Responses with example values of their preferred media type.                                                        |

## Schema UI

`@fumadocs/api-docs/headless` generates the data of Schema UI with `generateSchemaUI()`, and holds its navigation state: `<SchemaUIProvider />` owns the path of opened schemas, starting from the root.

```tsx title="components/api/schema/client.tsx"
'use client';
import {
  type SchemaUIGeneratedData,
  SchemaUIProvider,
  useSchemaTabs,
  useSchemaUI,
} from '@fumadocs/api-docs/headless';

export function SchemaUI({ name, generated }: { name: string; generated: SchemaUIGeneratedData }) {
  return (
    <SchemaUIProvider name={name} generated={generated}>
      <Body pathIndex={0} />
    </SchemaUIProvider>
  );
}

function Body({ pathIndex }: { pathIndex: number }) {
  const { path, open, generated } = useSchemaUI();
  const [selected, setSelected] = useSchemaTabs(pathIndex, 0);
  const schema = generated.refs[path[pathIndex].$ref];

  if (schema.type === 'or') {
    return (
      <select
        value={selected ?? schema.items[0].$type}
        onChange={(e) => setSelected(e.target.value)}
      >
        {schema.items.map((item) => (
          <option key={item.$type} value={item.$type}>
            {item.name}
          </option>
        ))}
      </select>
    );
  }

  if (schema.type === 'object') {
    return schema.props.map((prop) => (
      <button key={prop.name} onClick={() => open(prop.name, prop.$type)}>
        {prop.name}
      </button>
    ));
  }
}
```

| Hook                                  | Description                                                           |
| ------------------------------------- | --------------------------------------------------------------------- |
| `useSchemaUI()`                       | The generated schema data, the opened schemas, `open()` and `back()`. |
| `useSchemaTabs(pathIndex, depth)`     | The selected member of a union schema.                                |
| `useSchemaPopover(name, $ref)`        | Open state of the popover of a root property.                         |
| `useSchemaHighlight(pathIndex, name)` | Whether a property is highlighted by the link it was opened from.     |
| `useCopySchemaLink(name)`             | Copy the link to a property.                                          |
