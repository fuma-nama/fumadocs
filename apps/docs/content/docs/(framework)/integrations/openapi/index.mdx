## Setup

Install the required packages.

```npm
npm i fumadocs-openapi shiki
```

### Generate Styles

Add the following line:

```css title="Tailwind CSS"
@import 'tailwindcss';
@import 'fumadocs-ui/css/neutral.css';
@import 'fumadocs-ui/css/preset.css';
/* [!code ++] */
@import 'fumadocs-openapi/css/preset.css';
```

### Configure Plugin

Create the OpenAPI server instance & `<OpenAPIPage />` component.

```ts tab="lib/openapi.ts"
import { createOpenAPI } from 'fumadocs-openapi/server';

// note: this is a server-side API
export const openapi = createOpenAPI({
  // the OpenAPI schema, you can also give it an external URL.
  input: ['./openapi.json'],
});
```

```tsx tab="components/api-page.tsx"
'use client';
import { createOpenAPIPage } from 'fumadocs-openapi/ui';

export const OpenAPIPage = createOpenAPIPage();
```

See [`createOpenAPI()`](/docs/integrations/openapi/server) & [`createOpenAPIPage()`](/docs/integrations/openapi/api-page) for available options.

### Generate Pages

Integrate into [Loader API](/docs/headless/source-api/source) to generate pages, without generating real files.

```ts tab="Fumadocs MDX" title="lib/source.ts"
import { loader } from 'fumadocs-core/source';
import { docs } from 'collections/server';
import { openapi } from '@/lib/openapi';

export const source = loader(
  // [!code ++:6]
  {
    docs: docs.toFumadocsSource(),
    openapi: await openapi.staticSource({
      baseDir: 'openapi',
    }),
  },
  {
    baseUrl: '/docs',
    // [!code ++] optional: adds a badge to each page item in page tree
    plugins: [openapi.loaderPlugin()],
    // ...
  },
);
```

```ts tab="Astro" title="src/lib/source.ts"
import { getCollection } from 'astro:content';
import { loader } from 'fumadocs-core/source';
import { toFumadocsSource } from '@fumadocs/astro/source';
import { openapi } from '@/lib/openapi';

export const source = loader(
  // [!code ++:6]
  {
    docs: toFumadocsSource(await getCollection('docs'), await getCollection('meta')),
    openapi: await openapi.staticSource({
      baseDir: 'openapi',
    }),
  },
  {
    baseUrl: '/docs',
    // [!code ++] optional: adds a badge to each page item in page tree
    plugins: [openapi.loaderPlugin()],
  },
);
```

It generates a page per operation by default, see [Generating Pages](/docs/integrations/openapi/generate-files) to customize them.

**It will change the type of your pages**, make sure to update all references to your `source`.

For example, where you return text for LLM:

```ts
import { llms } from 'fumadocs-core/source';
import { source } from '@/lib/source';

export const docsLlms = llms(source, {
  renderPage: (page) => {
    // [!code ++:4]
    if (page.type === 'openapi') {
      // e.g. return the stringified OpenAPI schema
      return JSON.stringify(page.data.getSchema().bundled, null, 2);
    }

    // your original flow below...
  },
});
```

Pass a client payload from server, then render the page using the `<OpenAPIPage />` component you created above.

For example, in Tanstack Start. Only import `source` on the server, `staticSource()` reads files: define [Macro API](/docs/mdx/macro) collections in another module, like `lib/collections.ts`.

```tsx title="routes/docs/$.tsx"
import { createFileRoute, notFound } from '@tanstack/react-router';
import { DocsLayout } from 'fumadocs-ui/layouts/docs';
import { createServerFn } from '@tanstack/react-start';
import { docs } from '@/lib/collections';
import { source } from '@/lib/source';
import { DocsBody, DocsPage, DocsTitle } from 'fumadocs-ui/layouts/docs/page';
import { useFumadocsLoader } from 'fumadocs-core/source/client';
import { type ReactNode, Suspense } from 'react';
import { OpenAPIPage } from '@/components/api-page';

export const Route = createFileRoute('/docs/$')({
  component: Page,
  loader: async ({ params }) => {
    const slugs = params._splat?.split('/') ?? [];
    const data = await serverLoader({ data: slugs });

    // Fumadocs MDX: only preload content for normal pages [!code highlight:3]
    if (data.type === 'docs') {
      await docs.getPage(data.path)?.preload();
    }
    return data;
  },
});

const serverLoader = createServerFn({
  method: 'GET',
})
  .validator((slugs: string[]) => slugs)
  .handler(async ({ data: slugs }) => {
    const page = source.getPage(slugs);
    if (!page) throw notFound();

    const pageTree = await source.serializePageTree(source.getPageTree());
    // different result for OpenAPI pages [!code ++:8]
    if (page.type === 'openapi') {
      return {
        type: 'openapi',
        title: page.data.title,
        pageTree,
        props: page.data.getOpenAPIPageProps(),
      };
    }

    return {
      type: 'docs',
      path: page.path,
      pageTree,
      // ...
    };
  });

// renders a page of your `docs` collection
function Content({ path }: { path: string }) {
  // ...
}

function Page() {
  const page = useFumadocsLoader(Route.useLoaderData());
  let content: ReactNode;

  // render OpenAPI page content [!code ++:10]
  if (page.type === 'openapi') {
    content = (
      <DocsPage full>
        <DocsTitle>{page.title}</DocsTitle>
        <DocsBody>
          {/* pass the payload data */}
          <OpenAPIPage {...page.props} />
        </DocsBody>
      </DocsPage>
    );
  } else {
    content = <Content path={page.path} />;
  }

  return (
    <DocsLayout tree={page.pageTree}>
      <Suspense>{content}</Suspense>
    </DocsLayout>
  );
}
```

You can see the full [Tanstack Start example](https://github.com/fuma-nama/fumadocs/blob/dev/examples/tanstack-start-openapi).

Pass the props of OpenAPI pages to the [`Docs` island](/docs/manual-installation/astro#docs-page), it renders `<OpenAPIPage />` in place of the content:

```tsx title="src/components/docs.tsx"
// [!code highlight]
import { DocsBody, DocsDescription, DocsPage, DocsTitle } from 'fumadocs-ui/layouts/docs/page';
// [!code ++:2]
import type { OpenAPIPageProps } from 'fumadocs-openapi/ui';
import { OpenAPIPage } from '@/components/api-page';

export interface DocsProps extends ProviderProps {
  // ...
  // [!code ++]
  openapi?: OpenAPIPageProps;
}

// [!code highlight]
export function Docs({ tree, title, description, toc, openapi, children, ...props }: DocsProps) {
  const data = useFumadocsLoader({ tree });

  return (
    <Provider {...props}>
      <DocsLayout {...baseOptions()} tree={data.tree}>
        {/* [!code ++:8] */}
        {openapi ? (
          <DocsPage toc={toc} full>
            <DocsTitle>{title}</DocsTitle>
            <DocsBody>
              <OpenAPIPage {...openapi} />
            </DocsBody>
          </DocsPage>
        ) : (
          <DocsPage toc={toc}>
            <DocsTitle>{title}</DocsTitle>
            <DocsDescription>{description}</DocsDescription>
            {children}
          </DocsPage>
        )}
      </DocsLayout>
    </Provider>
  );
}
```

```astro title="src/pages/docs/[...slug].astro"
---
// ...
const { page } = Astro.props;
// [!code highlight]
const { Content, toc } = page.type === 'docs' ? await page.data.render() : { toc: page.data.toc };
// [!code ++]
const openapi = page.type === 'openapi' ? page.data.getOpenAPIPageProps() : undefined;
---

    params={Astro.params}
    tree={pageTree}
    title={page.data.title}
    description={page.data.description}
    toc={toc}
    openapi={openapi}
  >
    <!-- [!code highlight] -->
    {Content && <Content />}
  </Docs>
</Layout>
```

## Features

The official OpenAPI integration supports:

- Basic API endpoint information
- Interactive API playground
- Example code to send request (in different programming languages)
- Response samples and TypeScript definitions
- Request parameters and body generated from schemas
