## Setup

Define the i18n configurations in a file, we will import it with `@/lib/i18n` in this guide.

```ts title="src/lib/i18n.ts"
import { defineI18n } from 'fumadocs-core/i18n';

export const i18n = defineI18n({
  defaultLanguage: 'en',
  languages: ['cn', 'en'],
});
```

> See [available options](/docs/headless/internationalization/config) for i18n config.

### Routing

Add a `[...lang]` prefix to your pages. The rest parameter lets the default language have no prefix (e.g. `/docs`).

<Files>
  <Folder name="src/pages" defaultOpen>
    <Folder name="[...lang]" defaultOpen>
      <File name="index.astro" />
      <Folder name="docs" defaultOpen>
        <File name="[...slug].astro" />
      </Folder>
    </Folder>
    <Folder name="api" defaultOpen>
      <File name="search.ts" />
    </Folder>
  </Folder>
</Files>

Enable `hideLocale`, so the URLs generated by Fumadocs match these routes:

```ts title="src/lib/i18n.ts"
export const i18n = defineI18n({
  defaultLanguage: 'en',
  languages: ['cn', 'en'],
  hideLocale: 'default-locale', // [!code ++]
});
```

### Translations

Define your translations in `lib/layout.shared.tsx`, see [Translations](/docs/ui/translations) for more details (including official language packs).

You can also add `locale` parameter to `baseOptions()` for localized layout options:

```tsx title="src/lib/layout.shared.tsx"
// [!code ++:2]
import { i18n } from '@/lib/i18n';
import { uiTranslations } from 'fumadocs-ui/i18n';
import type { BaseLayoutProps } from 'fumadocs-ui/layouts/shared';

// [!code ++:3]
export const translations = i18n
  .translations()
  .extend(uiTranslations());

export function baseOptions(): BaseLayoutProps { // [!code --]
export function baseOptions(locale: string): BaseLayoutProps { // [!code ++]
  return {
    nav: {
      title: 'My App',
      url: locale === i18n.defaultLanguage ? '/' : `/${locale}`, // [!code ++]
    },
  };
}
```

Pass translations to `<Provider />`. The MDX content is an island of its own, so it needs a provider too:

```tsx tab="src/components/provider.tsx"
import { navigate } from 'astro:transitions/client';
import type { ReactNode } from 'react';
import { RootProvider } from 'fumadocs-ui/provider/astro';
// [!code ++:2]
import { i18nProvider } from 'fumadocs-ui/i18n';
import { translations } from '@/lib/layout.shared';
import SearchDialog from '@/components/search';

export interface ProviderProps {
  lang: string; // [!code ++]
  pathname: string;
  params: Record<string, string | undefined>;
  children?: ReactNode;
}

export function Provider({ lang, pathname, params, children }: ProviderProps) {
  return (
    <RootProvider
      pathname={pathname}
      params={params}
      navigate={navigate}
      // [!code ++]
      i18n={i18nProvider(translations, lang)}
      search={{ SearchDialog }}
    >
      {children}
    </RootProvider>
  );
}

// [!code ++:16]
/**
 * For the MDX content, which is an island of its own.
 */
export function ContentProvider({ lang, pathname, params, children }: ProviderProps) {
  return (
    <RootProvider
      pathname={pathname}
      params={params}
      navigate={navigate}
      i18n={i18nProvider(translations, lang)}
      theme={{ enabled: false }}
      search={{ enabled: false }}
    >
      {children}
    </RootProvider>
  );
}
```

```tsx tab="src/components/mdx.tsx"
import defaultMdxComponents from 'fumadocs-ui/mdx';
import { DocsBody } from 'fumadocs-ui/layouts/docs/page';
import type { MDXComponents } from 'mdx/types';
import { ContentProvider, type ProviderProps } from '@/components/provider'; // [!code ++]

export function getMDXComponents(components?: MDXComponents) {
  return {
    ...defaultMdxComponents,
    ...components,
  } satisfies MDXComponents;
}

// for the Fumadocs integration, `wrapper` receives the props of `<Content />`
export default getMDXComponents({
  wrapper: DocsBody, // [!code --]
  // [!code ++:5]
  wrapper: ({ children, ...props }: ProviderProps) => (
    <ContentProvider {...props}>
      <DocsBody>{children}</DocsBody>
    </ContentProvider>
  ),
});
```

### Pass Locale

Pass the locale to Fumadocs in your source, layouts and pages.

```ts tab="src/lib/source.ts"
import { i18n } from '@/lib/i18n'; // [!code ++]

export const source = loader({
  source: toFumadocsSource(await getCollection('docs'), await getCollection('meta')),
  baseUrl: '/docs',
  i18n, // [!code ++]
});
```

```tsx tab="src/components/docs.tsx"
export function Docs({ tree, title, description, toc, children, ...props }: DocsProps) {
  const data = useFumadocsLoader({ tree });

  return (
    <Provider {...props}>
      {/* [!code highlight] */}
      <DocsLayout {...baseOptions(props.lang)} tree={data.tree}>
        {/* ... */}
      </DocsLayout>
    </Provider>
  );
}
```

```astro tab="src/layouts/Layout.astro"
---
interface Props {
  lang: string; // [!code ++]
  title: string;
  description?: string;
}

const { lang, title, description } = Astro.props;
---

<html lang={lang} transition:animate="none">
  {/* ... */}
</html>
```

```astro tab="Docs Page" title="src/pages/[...lang]/docs/[...slug].astro"
---
import type { GetStaticPaths, InferGetStaticPropsType } from 'astro';
import Layout from '@/layouts/Layout.astro';
import { Docs } from '@/components/docs';
import { i18n } from '@/lib/i18n';
import { source } from '@/lib/source';

export const prerender = true;

export const getStaticPaths = (() => {
  const paths = [];
  for (const { language, pages } of source.getLanguages()) {
    for (const page of pages) {
      paths.push({
        params: {
          lang: language === i18n.defaultLanguage ? undefined : language,
          slug: decodeURI(page.slugs.join('/')) || undefined,
        },
        props: { lang: language, page },
      });
    }
  }
  return paths;
}) satisfies GetStaticPaths;

type Props = InferGetStaticPropsType<typeof getStaticPaths>;

const { lang, page } = Astro.props;
const { body: Content, toc } = await page.data.load();
const pageTree = await source.serializePageTree(source.getPageTree(lang));
const provider = { lang, pathname: Astro.url.pathname, params: Astro.params };
---

<Layout lang={lang} title={page.data.title} description={page.data.description}>
  <Docs
    client:load
    transition:persist="docs"
    {...provider}
    tree={pageTree}
    title={page.data.title}
    description={page.data.description}
    toc={toc}
  >
    <Content {...provider} />
  </Docs>
</Layout>
```

```astro tab="Home Page" title="src/pages/[...lang]/index.astro"
---
import type { GetStaticPaths, InferGetStaticPropsType } from 'astro';
import Layout from '@/layouts/Layout.astro';
import { i18n } from '@/lib/i18n';

export const prerender = true;

export const getStaticPaths = (() => {
  return i18n.languages.map((lang) => ({
    params: { lang: lang === i18n.defaultLanguage ? undefined : lang },
    props: { lang },
  }));
}) satisfies GetStaticPaths;

type Props = InferGetStaticPropsType<typeof getStaticPaths>;

const { lang } = Astro.props;
---

<Layout lang={lang} title="My App">
  <a href={lang === i18n.defaultLanguage ? '/docs' : `/${lang}/docs`}>
    Open Docs
  </a>
</Layout>
```

### Navigation

Fumadocs only handles navigation for its own layouts (e.g. sidebar).
For other places, extract the locale from `Astro.params` or the path and prepend it to links.

```astro
---
const { lang } = Astro.params;
---

<a href={lang ? `/${lang}/about` : '/about'}>About Us</a>
```

In addition, the [`fumadocs-core/dynamic-link`](/docs/headless/components/link#dynamic-hrefs) component supports dynamic hrefs, you can use it to prepend the locale prefix.
It is useful for Markdown/MDX content. Add it to your MDX components, the MDX content renders them inside its own island:

```tsx title="src/components/mdx.tsx"
import { DynamicLink } from 'fumadocs-core/dynamic-link'; // [!code ++]

export function getMDXComponents(components?: MDXComponents) {
  return {
    ...defaultMdxComponents,
    DynamicLink, // [!code ++]
    ...components,
  } satisfies MDXComponents;
}
```

```mdx title="content.mdx"
<DynamicLink href="/[lang]/another-page">This is a link</DynamicLink>
```
