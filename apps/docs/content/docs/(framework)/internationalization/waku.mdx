## Setup

Define the i18n configurations in a file, we will import it with `@/lib/i18n` in this guide.

```ts title="src/lib/i18n.ts"
import { defineI18n } from 'fumadocs-core/i18n';

export const i18n = defineI18n({
  defaultLanguage: 'en',
  languages: ['cn', 'en'],
});
```

> See [available options](/docs/headless/internationalization/config) for i18n config.

### Routing

Add a `[lang]` prefix to all your pages.

### Translations

Define your translations in `lib/layout.shared.tsx`, see [Translations](/docs/ui/translations) for more details (including official language packs).

You can also add `locale` parameter to `baseOptions()` for localized layout options:

```tsx title="src/lib/layout.shared.tsx"
// [!code ++:2]
import { i18n } from '@/lib/i18n';
import { uiTranslations } from 'fumadocs-ui/i18n';
import type { BaseLayoutProps } from 'fumadocs-ui/layouts/shared';

// [!code ++:11]
export const translations = i18n
  .translations()
  .extend(uiTranslations())
  .add({
    en: {
      displayName: 'English',
    },
    cn: {
      displayName: 'Chinese',
    },
  });

export function baseOptions(): BaseLayoutProps { // [!code --]
export function baseOptions(locale: string): BaseLayoutProps { // [!code ++]
  return {
    nav: {
      title: 'My App',
      url: `/${locale}`, // [!code ++]
    },
  };
}
```

Pass translations to `<RootProvider />`:

```tsx title="src/components/provider.tsx"
'use client';
import type { ReactNode } from 'react';
import { RootProvider } from 'fumadocs-ui/provider/waku';
// [!code ++:4]
import { i18nProvider } from 'fumadocs-ui/i18n';
import { useRouter } from 'waku/router/client';
import { i18n } from '@/lib/i18n';
import { translations } from '@/lib/layout.shared';

export function Provider({ children }: { children: ReactNode }) {
  // the first segment of path [!code ++:2]
  const lang = useRouter().path.split('/')[1] || i18n.defaultLanguage;

  return <RootProvider>{children}</RootProvider>; // [!code --]
  return <RootProvider i18n={i18nProvider(translations, lang)}>{children}</RootProvider>; // [!code ++]
}
```

### Pass Locale

Pass the locale to Fumadocs in your pages and layouts, layouts and pages under `[lang]` receive it as a prop.

```ts tab="src/lib/source.ts"
import { i18n } from '@/lib/i18n'; // [!code ++]

export const source = loader({
  source: docs.toFumadocsSource(),
  baseUrl: '/docs',
  i18n, // [!code ++]
});
```

```tsx tab="Home Layout" title="src/pages/[lang]/(home)/_layout.tsx"
export default function Layout({ children }: { children: ReactNode }) { // [!code --]
export default function Layout({ lang, children }: { lang: string; children: ReactNode }) { // [!code ++]
  return <HomeLayout {...baseOptions()}>{children}</HomeLayout>; // [!code --]
  return <HomeLayout {...baseOptions(lang)}>{children}</HomeLayout>; // [!code ++]
}
```

```tsx tab="Home Page" title="src/pages/[lang]/(home)/index.tsx"
import { i18n } from '@/lib/i18n'; // [!code ++]

// ...

export async function getConfig() {
  return {
    render: 'static',
    staticPaths: i18n.languages, // [!code ++]
  };
}
```

```tsx tab="Docs Layout" title="src/pages/[lang]/docs/_layout.tsx"
export default function Layout({ children }: { children: ReactNode }) { // [!code --]
export default function Layout({ lang, children }: { lang: string; children: ReactNode }) { // [!code ++]
  return (
    // [!code highlight]
    <DocsLayout {...baseOptions(lang)} tree={source.getPageTree(lang)}>
      {children}
    </DocsLayout>
  );
}
```

```tsx tab="Docs Page" title="src/pages/[lang]/docs/[...slugs].tsx"
export default function Page({ slugs }: PageProps<'/docs/[...slugs]'>) { // [!code --]
export default function Page({ lang, slugs }: PageProps<'/[lang]/docs/[...slugs]'>) { // [!code ++]
  const page = source.getPage(slugs); // [!code --]
  const page = source.getPage(slugs, lang); // [!code ++]
  if (!page) unstable_notFound();

  // ...
}

export async function getConfig() {
  const pages = source
    .generateParams()
    .map((item) => (item.lang ? [item.lang, ...item.slug] : item.slug)); // [!code highlight]

  return {
    render: 'static',
    staticPaths: pages,
  } as const;
}
```

### Navigation

Fumadocs only handles navigation for its own layouts (e.g. sidebar).
For other places, extract the locale from the path and prepend it to links.

```tsx
import { Link, useRouter } from 'waku/router/client';

const router = useRouter();
const lang = router.path.split('/')[1];

<Link to={`/${lang}/about`}>About Us</Link>;
```

In addition, the [`fumadocs-core/dynamic-link`](/docs/headless/components/link#dynamic-hrefs) component supports dynamic hrefs, you can use it to prepend the locale prefix.
It is useful for Markdown/MDX content.

```mdx title="content.mdx"
import { DynamicLink } from 'fumadocs-core/dynamic-link';

<DynamicLink href="/[lang]/another-page">This is a link</DynamicLink>
```
