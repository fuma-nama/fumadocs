## Setup

Define the i18n configurations in a file, we will import it with `@/lib/i18n` in this guide.

```ts title="app/lib/i18n.ts"
import { defineI18n } from 'fumadocs-core/i18n';

export const i18n = defineI18n({
  defaultLanguage: 'en',
  languages: ['cn', 'en'],
});
```

> See [available options](/docs/headless/internationalization/config) for i18n config.

### Routing

Add a `:lang` prefix to your pages:

```ts title="app/routes.ts"
import { index, route, type RouteConfig } from '@react-router/dev/routes'; // [!code --]
import { route, type RouteConfig } from '@react-router/dev/routes'; // [!code ++]

export default [
  index('routes/home.tsx'), // [!code --]
  route('docs/*', 'routes/docs.tsx'), // [!code --]
  // [!code ++:2]
  route(':lang', 'routes/home.tsx'),
  route(':lang/docs/*', 'routes/docs.tsx'),
  route('api/search', 'routes/search.ts'),
] satisfies RouteConfig;
```

To serve the default language without a prefix (e.g. `/docs`), make the parameter optional and enable `hideLocale`:

```ts tab="app/routes.ts"
export default [
  // [!code highlight:2]
  route(':lang?', 'routes/home.tsx'),
  route(':lang?/docs/*', 'routes/docs.tsx'),
  route('api/search', 'routes/search.ts'),
] satisfies RouteConfig;
```

```ts tab="app/lib/i18n.ts"
export const i18n = defineI18n({
  defaultLanguage: 'en',
  languages: ['cn', 'en'],
  hideLocale: 'default-locale', // [!code ++]
});
```

### Translations

Define your translations in `lib/layout.shared.tsx`, see [Translations](/docs/ui/translations) for more details (including official language packs).

You can also add `locale` parameter to `baseOptions()` for localized layout options:

```tsx title="app/lib/layout.shared.tsx"
// [!code ++:2]
import { i18n } from '@/lib/i18n';
import { uiTranslations } from 'fumadocs-ui/i18n';
import type { BaseLayoutProps } from 'fumadocs-ui/layouts/shared';

// [!code ++:11]
export const translations = i18n
  .translations()
  .extend(uiTranslations())
  .add({
    en: {
      displayName: 'English',
    },
    cn: {
      displayName: 'Chinese',
    },
  });

export function baseOptions(): BaseLayoutProps { // [!code --]
export function baseOptions(locale: string): BaseLayoutProps { // [!code ++]
  return {
    // different props based on `locale`
  };
}
```

Pass translations to `<RootProvider />`:

```tsx title="app/root.tsx"
import {
  isRouteErrorResponse,
  Links,
  Meta,
  Outlet,
  Scripts,
  ScrollRestoration,
  useParams, // [!code ++]
} from 'react-router';
import { RootProvider } from 'fumadocs-ui/provider/react-router';
// [!code ++:3]
import { i18nProvider } from 'fumadocs-ui/i18n';
import { i18n } from '@/lib/i18n';
import { translations } from '@/lib/layout.shared';

export function Layout({ children }: { children: React.ReactNode }) {
  const { lang = i18n.defaultLanguage } = useParams(); // [!code ++]

  return (
    // [!code highlight]
    <html lang={lang} suppressHydrationWarning>
      <head>{/* ... */}</head>
      <body className="flex flex-col min-h-screen">
        <RootProvider
          // [!code ++]
          i18n={i18nProvider(translations, lang)}
        >
          {children}
        </RootProvider>
        <ScrollRestoration />
        <Scripts />
      </body>
    </html>
  );
}
```

### Pass Locale

Pass the locale to Fumadocs in your pages and layouts.

```ts tab="app/lib/source.ts"
import { i18n } from '@/lib/i18n'; // [!code ++]

export const source = loader({
  source: docs.toFumadocsSource(),
  baseUrl: '/docs',
  i18n, // [!code ++]
});
```

```tsx tab="Home Page" title="app/routes/home.tsx"
export default function Home() { // [!code --]
export default function Home({ params }: Route.ComponentProps) { // [!code ++]
  return (
    // [!code highlight]
    <HomeLayout {...baseOptions(params.lang)}>
      {/* ... */}
    </HomeLayout>
  );
}
```

```tsx tab="Docs Page" title="app/routes/docs.tsx"
export async function loader({ params }: Route.LoaderArgs) {
  const slugs = params['*'].split('/').filter((v) => v.length > 0);
  const page = source.getPage(slugs); // [!code --]
  const page = source.getPage(slugs, params.lang); // [!code ++]
  if (!page) throw new Response('Not found', { status: 404 });

  return {
    path: page.path,
    pageTree: await source.serializePageTree(source.getPageTree()), // [!code --]
    pageTree: await source.serializePageTree(source.getPageTree(params.lang)), // [!code ++]
  };
}

export default function Page({ loaderData }: Route.ComponentProps) { // [!code --]
export default function Page({ loaderData, params }: Route.ComponentProps) { // [!code ++]
  const { path, pageTree } = useFumadocsLoader(loaderData);

  return (
    // [!code highlight]

      <Content path={path} />
    </DocsLayout>

}
```

### Navigation

Fumadocs only handles navigation for its own layouts (e.g. sidebar).
For other places, you can use the `useParams` hook to get the locale from url.

```tsx
import { Link, useParams } from 'react-router';

const { lang } = useParams();

<Link to={`/${lang}/about`}>About Us</Link>;
```

In addition, the [`fumadocs-core/dynamic-link`](/docs/headless/components/link#dynamic-hrefs) component supports dynamic hrefs, you can use it to prepend the locale prefix.
It is useful for Markdown/MDX content.

```mdx title="content.mdx"
import { DynamicLink } from 'fumadocs-core/dynamic-link';

<DynamicLink href="/[lang]/another-page">This is a link</DynamicLink>
```
